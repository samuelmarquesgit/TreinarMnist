Exemplo de teste HTML com pytest + html5lib

# tests/test_html_docs.py
import pytest, html5lib
from pathlib import Path
from bs4 import BeautifulSoup

HTML_DOCS = list(Path("docs").glob("*.html"))

@pytest.mark.parametrize("caminho", HTML_DOCS)
def test_html_valido(caminho):
    conteudo = caminho.read_text(encoding="utf-8")
    parser = html5lib.HTMLParser(strict=True)
    parser.parseFragment(conteudo)  # lança ParseError se inválido

@pytest.mark.parametrize("caminho", HTML_DOCS)
def test_html_tem_titulo(caminho):
    soup = BeautifulSoup(caminho.read_bytes(), "html5lib")
    assert soup.find("title") or soup.find("h1")
